class FakeProcess:
    def __init__(self, pid=9876, exit_after_signal=0, kill_exit=-9, never_reap=False):
        self.pid = pid
        self._code = None
        self.exit_after_signal = exit_after_signal
        self.kill_exit = kill_exit
        self.never_reap = never_reap
        self.signals = []
        self.killed = False
    def poll(self):
        return self._code
    def wait(self, timeout=None):
        if self.never_reap:
            raise subprocess.TimeoutExpired('fake', timeout)
        if self._code is None:
            if self.killed:
                self._code = self.kill_exit
            elif self.signals:
                self._code = self.exit_after_signal
            else:
                raise subprocess.TimeoutExpired('fake', timeout)
        return self._code
    def send_signal(self, sig):
        self.signals.append(sig)
        if not self.never_reap:
            self._code = self.exit_after_signal
    def kill(self):
        self.killed = True

    def _tool_observation(self, root, rustc_link):
        target = root / 'rustup-real'
        target.write_text('#!/bin/sh\nexit 0\n')
        target.chmod(0o700)
        expected_rustc = root / 'rustc-expected'
        expected_rustc.symlink_to(target)
        cargo = root / 'cargo'; cargo.write_text('cargo'); cargo.chmod(0o700)
        nextest = root / 'cargo-nextest'; nextest.write_text('nextest'); nextest.chmod(0o700)
        cc = root / 'cc'; cc.write_text('cc'); cc.chmod(0o700)
        clang = root / 'clang'; clang.write_text('clang'); clang.chmod(0o700)
        ld = root / 'ld'; ld.write_text('ld'); ld.chmod(0o700)
        lookups = {'cargo':str(cargo), 'rustc':str(rustc_link),
                   'cargo-nextest':str(nextest), 'cc':str(cc),
                   'clang':str(clang), 'ld':str(ld)}
        def record(path):
            return {'path':str(path), 'sha256':mod._sha256_file(path, time.monotonic_ns()+30_000_000_000)}
        expected = {
            'host': {'system':mod.platform.system(), 'machine':mod.platform.machine(), 'release':mod.platform.release()},
            'invocation_tools': {'python':record(Path(mod.sys.executable).resolve()),
                'cargo':record(cargo), 'rustc_path_entry':record(expected_rustc),
                'cargo_nextest':record(nextest)},
            'selected_toolchain': {'cc':record(cc), 'clang':record(clang), 'linker':record(ld)},
        }
        return expected, lookups

    def _verify_lookups(self, expected, lookups):
        with patch.object(mod.subprocess, 'run', side_effect=lambda argv, **kwargs:
                subprocess.CompletedProcess(argv, 0, lookups[argv[1]], '')):
            return mod.verify_selected_tools(expected, time.monotonic_ns()+30_000_000_000, {'PATH':'synthetic'})

    def test_wait_observer_exception_waits_for_real_three_second_term_cleanup(self):
        with tempfile.TemporaryDirectory() as td:
            root=Path(td); marker=root/'ready'; log=root/'log'; holder={}
            code=('import pathlib,signal,sys,time; marker=pathlib.Path(sys.argv[1]); '
                  'signal.signal(signal.SIGTERM,lambda *_:(time.sleep(3),sys.exit(0))); '
                  'marker.write_text("ready"); print("synthetic-ready",flush=True); signal.pause()')
            command=[str(Path(mod.sys.executable)),'-c',code,str(marker)]
            class WaitFault:
                def __init__(self, inner): self.inner=inner; self.raised=False
                @property
                def pid(self): return self.inner.pid
                def poll(self): return self.inner.poll()
                def wait(self, timeout=None):
                    if not self.raised:
                        self.raised=True
                        raise RuntimeError('synthetic wait observer fault')
                    return self.inner.wait(timeout=timeout)
                def send_signal(self,sig): return self.inner.send_signal(sig)
                def kill(self): return self.inner.kill()
            def factory(argv,**kwargs):
                proc=subprocess.Popen(argv,**kwargs); holder['proc']=proc
                limit=time.monotonic()+5
                while time.monotonic()<limit and not marker.exists(): time.sleep(.01)
                if not marker.exists():
                    proc.kill(); proc.wait(timeout=2)
                    raise AssertionError('synthetic child failed readiness')
                return WaitFault(proc)
            began=time.monotonic()
            event=mod.run_supervised(command,log_path=log,deadline_ns=time.monotonic_ns()+30_000_000_000,
                env={'PATH':os.environ.get('PATH','/usr/bin:/bin'),'LANG':'C'},worker_timeout=10,popen_factory=factory)
            elapsed=time.monotonic()-began
            self.assertGreaterEqual(elapsed,2.8)
            self.assertIn('supervision_exception',event)
            self.assertEqual(event['supervision_exception']['type'],'RuntimeError')
            self.assertTrue(event['direct_worker_reaped'])
            self.assertEqual(event['exit_status'],0)
            self.assertFalse(event['success'])
            self.assertIsNotNone(holder['proc'].poll())
            self.assertIn('synthetic-ready',log.read_text())
